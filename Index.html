Custom stats table — skew, kurtosis, normality, missing%
python
def deep_stats(df):
    rows = []
    for col in df.select_dtypes('number').columns:
        d = df[col].dropna()
        _, p_norm = shapiro(d.sample(min(5000, len(d))))
        rows.append({
            'col': col, 'mean': d.mean(), 'median': d.median(),
            'std': d.std(), 'skew': skew(d), 'kurt': kurtosis(d),
            'q1': d.quantile(.25), 'q3': d.quantile(.75),
            'normal': p_norm > 0.05,
            'miss%': df[col].isna().mean() * 100
        })
    return pd.DataFrame(rows).set_index('col')

deep_stats(df)

# IQR method (per column)
def iqr_flag(df, col):
    Q1, Q3 = df[col].quantile([.25, .75])
    IQR = Q3 - Q1
    return (df[col] < Q1-1.5*IQR) | (df[col] > Q3+1.5*IQR)

for col in df.select_dtypes('number').columns:
    n = iqr_flag(df, col).sum()
    print(f"{col}: {n} outliers ({n/len(df)*100:.1f}%)")

# Z-score: remove rows with |z| > 3
from scipy.stats import zscore
z = df.select_dtypes('number').apply(zscore)
df_clean = df[(z.abs() < 3).all(axis=1)]
print(f"Kept {len(df_clean)}/{len(df)} rows")

Memory Optimisation

downcast dtypes, categorise strings

Auto-reduce memory — downcast all numeric + categorise low-cardinality
python
def reduce_memory(df):
    before = df.memory_usage(deep=True).sum() / 1e6
    for col in df.select_dtypes('int').columns:
        df[col] = pd.to_numeric(df[col], downcast='integer')
    for col in df.select_dtypes('float').columns:
        df[col] = pd.to_numeric(df[col], downcast='float')
    for col in df.select_dtypes('object').columns:
        if df[col].nunique() / len(df) < 0.5:
            df[col] = df[col].astype('category')
    after = df.memory_usage(deep=True).sum() / 1e6
    print(f"{before:.2f} MB → {after:.2f} MB ({(1-after/before)*100:.0f}% saved)")
    return df

df = reduce_memory(df)

Effect Size & Power

Cohen's d, minimum sample size

Cohen's d + required n for 80% power
python
from statsmodels.stats.power import TTestIndPower
def cohens_d(a,b):
    return abs(a.mean()-b.mean()) / np.sqrt((a.std()**2+b.std()**2)/2)
d = cohens_d(df['g_a'],df['g_b'])
print(f"Cohen's d={d:.3f}") # 0.2=small 0.5=med 0.8=large
n = TTestIndPower().solve_power(
    effect_size=d, alpha=0.05, power=0.80)
print(f"Required n per group: {n:.0f}")

from statsmodels.tsa.stattools import adfuller, kpss
from statsmodels.tsa.seasonal import STL

def stationarity_check(series):
    adf_p  = adfuller(series.dropna())[1]
    kpss_p = kpss(series.dropna())[1]
    print(f"ADF  p={adf_p:.4f}  {'Stationary✓' if adf_p<0.05 else 'Non-stationary✗'}")

stationarity_check(df['value'])
# Make stationary
df['diff1']   = df['value'].diff()

Build Cohort Table

cohort month, period index, pivot

Assign cohort month and period number, pivot to matrix
python
# Required columns: user_id, order_date (datetime)
df['order_month'] = df['order_date'].dt.to_period('M')
# First month each user appeared
df['cohort'] = df.groupby('user_id')['order_month'].transform('min')
# Months since first purchase
df['period'] = (df['order_month']-df['cohort']).apply(lambda x:x.n)
# Pivot: rows=cohort, cols=period, values=unique users
pivot = df.groupby(['cohort','period'])['user_id'].\
           nunique().reset_index().\
           pivot(index='cohort',columns='period',values='user_id')
print(pivot.head())

cohort_size = pivot.iloc[:, 0]

Revenue LTV per Cohort

cumulative per-user LTV curves

Cumulative revenue per user — compare cohort LTV over time
python
rev_pivot = df.groupby(['cohort','period'])['revenue'].\
              sum().reset_index().\
              pivot(index='cohort',columns='period',values='revenue')
# Per-user normalisation
ltv = rev_pivot.divide(cohort_size, axis=0).cumsum(axis=1)
ltv.T.plot(figsize=(12,5),
    title='Cumulative LTV per User by Cohort', alpha=.8)
plt.xlabel('Period (months)'); plt.ylabel('Revenue / User')
plt.show()

Advanced
01

Kaplan-Meier Survival Curve

churn probability over time

KM curve + log-rank test to compare segments
python
from lifelines import KaplanMeierFitter  # pip install lifelines
from lifelines.statistics import logrank_test
# df needs: duration (months active), churned (1/0)
kmf = KaplanMeierFitter()
kmf.fit(df['duration'], event_observed=df['churned'])
kmf.plot_survival_function(ci_show=True)
plt.title('Kaplan-Meier Survival Curve'); plt.show()
# Compare two cohorts
g1=df[df['plan']=='basic']; g2=df[df['plan']=='premium']
res = logrank_test(g1['duration'],g2['duration'],
    g1['churned'],g2['churned'])
print(f"Log-rank p={res.p_value:.4f}")

Cox Proportional Hazards

feature impact on churn risk

CoxPH hazard ratios — which features drive churn
python
from lifelines import CoxPHFitter
cph = CoxPHFitter()
cph.fit(
    df[['duration','churned','age','plan','monthly_spend']],
    duration_col='duration', event_col='churned')
cph.print_summary()
cph.plot(); plt.show()  # hazard ratio forest plot
# Predict median survival for a new user
new_user = pd.DataFrame({'age':[28],'plan':['premium'],'monthly_spend':[49]})
